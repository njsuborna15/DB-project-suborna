CREATE TABLE Location (
    Location_ID INT PRIMARY KEY AUTO_INCREMENT,
    District VARCHAR(50) NOT NULL,
    Latitude DECIMAL(9,6) NOT NULL,
    Longitude DECIMAL(9,6) NOT NULL,
    Elevation DECIMAL(6,2) NOT NULL,
    Soil_Type VARCHAR(20) NOT NULL,
    Last_Assessed DATE
);

CREATE TABLE Landslide_Event (
    Event_ID INT PRIMARY KEY AUTO_INCREMENT,
    Location_ID INT,
    Event_Date DATE NOT NULL,
    Casualties INT,
    Rainfall DECIMAL(6,2),
    Description TEXT,
    FOREIGN KEY (Location_ID) REFERENCES Location(Location_ID)
);

CREATE TABLE Risk_Assessment (
    Assessment_ID INT PRIMARY KEY AUTO_INCREMENT,
    Location_ID INT,
    Assessment_Date DATE NOT NULL,
    Risk_Level ENUM('Low', 'Medium', 'High') NOT NULL,
    Recommendations TEXT,
    FOREIGN KEY (Location_ID) REFERENCES Location(Location_ID)
);

CREATE TABLE Mitigation_Measure (
    Measure_ID INT PRIMARY KEY AUTO_INCREMENT,
    Location_ID INT,
    Agency_ID INT,
    Measure_Name VARCHAR(100) NOT NULL,
    Description TEXT,
    Implementation_Date DATE,
    FOREIGN KEY (Location_ID) REFERENCES Location(Location_ID),
    FOREIGN KEY (Agency_ID) REFERENCES Government_Agency(Agency_ID)
);

CREATE TABLE Government_Agency (
    Agency_ID INT PRIMARY KEY AUTO_INCREMENT,
    Agency_Name VARCHAR(100) NOT NULL,
    Contact_Person VARCHAR(100) NOT NULL,
    Phone VARCHAR(20) NOT NULL,
    Email VARCHAR(100) NOT NULL
);
